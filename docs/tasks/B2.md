import json, glob, re
import mandarin_course.publish_timeline  # noqa: F401 (breaks an import cycle)

from pathlib import Path
v = load_vocab(Path("lessons/lesson_002_vocab.json"), "lesson_002"); known = {w.hanzi for w in v.words}
missing = set()
for f in glob.glob("lessons/lesson_002_*.json"):

print("missing", len(missing), sorted(missing)[:20])
for lid, srcs in {"lesson_001": ["lessons/lesson_001.json"], "lesson_002": [f for f in glob.glob("lessons/lesson_002_*.json") if not f.endswith(("_vocab.json", "_notes.json"))]}.items():

E
